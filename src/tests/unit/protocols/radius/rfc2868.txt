# Load libfreeradius-radius
proto radius
proto-dictionary radius
fuzzer-out radius

#
#  The salt is taken from fr_fast_rand(), with a seed set
#  in src/protocols/radius/encode.c, of { 6809, 2112 }
#
#  In normal operation, the seed is taken from fr_rand()
#
encode-pair Tunnel-Password = "foo"
match 45 15 00 99 73 05 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e

decode-pair -
match Tunnel-Password = "foo"

#
#  Note that the first tunnel password is encoded exactly the same as above.
#
#  The second one has it's own salt, which includes a counter to guarantee uniqueness.
#
encode-pair Tunnel-Password = "foo", Tunnel-Password = "bar"
match  45 15 00 99 73 05 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 45 15 00 9d c5 c4 69 23 3a 16 57 b3 5c 97 82 3c 97 ec 6b 7e f1

decode-pair -
match Tunnel-Password = "foo", Tunnel-Password = "bar"

#
#  A different password results in the salt staying the same, but a
#  different encoded password.  Note that the entire encoded string
#  is different.
#
encode-pair Tunnel-Password = "barbar"
match 45 15 00 c0 6f 25 5f 09 b7 dc 0e 4a 1a 46 56 05 f4 8f 7c 0b a4

decode-pair -
match Tunnel-Password = "barbar"

#
#  Tunnel-Password is automatically skipped when it's not possible to encode it securely.
#
encode-proto Packet-Type = Access-Request, User-Name = "bob", Tunnel-Password = "foo", User-Password = "bob"
match 01 00 00 2b 00 01 02 03 04 05 06 07 08 09 0a 0b 0c 0d 0e 0f 01 05 62 6f 62 02 12 f4 81 6b ca 74 fd 7a 1a 10 46 07 24 00 14 82 8b

decode-proto -
match Packet-Type = ::Access-Request, Packet-Authentication-Vector = 0x000102030405060708090a0b0c0d0e0f, User-Name = "bob", User-Password = "bob"


#
#  Integer attributes
#
encode-pair Tunnel-Type = PPTP
match 40 06 00 00 00 01

decode-pair -
match Tunnel-Type = ::PPTP

decode-pair 40 06 01 00 00 01
match Tag-1 = { Tunnel-Type = ::PPTP }

encode-pair -
match 40 06 01 00 00 01

#
#  Two of the same tags, they're grouped together.
#
decode-pair 40 06 01 00 00 01 41 06 01 00 00 01
match Tag-1 = { Tunnel-Type = ::PPTP, Tunnel-Medium-Type = ::IPv4 }

encode-pair -
match 40 06 01 00 00 01 41 06 01 00 00 01

#
# tag1, tag2, tag1 --> { tag1, tag1 }, tag2
#
decode-pair 40 06 01 00 00 01 40 06 02 00 00 01 41 06 01 00 00 01
match Tag-1 = { Tunnel-Type = ::PPTP, Tunnel-Medium-Type = ::IPv4 }, Tag-2 = { Tunnel-Type = ::PPTP }

#
#  The "encode" will group attributes by tag value, unlike the input packet.
#
encode-pair -
match 40 06 01 00 00 01 41 06 01 00 00 01 40 06 02 00 00 01

#
#  Tags are aggregated across the entire packet, too.
#
decode-proto 01 01 00 26 00000000000000000000000000000001 40 06 01 00 00 01 40 06 02 00 00 01 41 06 01 00 00 01
match Packet-Type = ::Access-Request, Packet-Authentication-Vector = 0x00000000000000000000000000000001, Tag-1 = { Tunnel-Type = ::PPTP, Tunnel-Medium-Type = ::IPv4 }, Tag-2 = { Tunnel-Type = ::PPTP }

#
#  and we encode the tags in order in the packet.
#
encode-proto -
match 01 00 00 26 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 01 40 06 01 00 00 01 41 06 01 00 00 01 40 06 02 00 00 01

# Invalid tag on decoding
decode-pair 40 06 20 00 00 01
match raw.Tunnel-Type = 0x20000001

#
#  String attributes
#
encode-pair Tunnel-Client-Endpoint = "foo"
match 42 05 66 6f 6f

decode-pair -
match Tunnel-Client-Endpoint = "foo"

#
# The first byte of the string looks like a tag.  In order to avoid
# confusion, we encode a zero tag.
#
encode-pair Tunnel-Client-Endpoint = "\001\002\003"
match 42 06 00 01 02 03

decode-pair -
match Tunnel-Client-Endpoint = "\001\002\003"

#
#  A tagged attribute which can't be decoded becomes a raw attribute at the top level.  It is
#  not left in its Tag-N group.  The other attributes in that group are still decoded.
#
#  This Tunnel-Password has a bad first byte of encrypted data, so the decrypted length is wrong.
#
decode-pair 45 15 01 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 40 06 01 00 00 01
match raw.Tunnel-Password = 0x019973851de6c557307dacd5daa599f4e26e7e, Tag-1 = { Tunnel-Type = ::PPTP }

decode-pair 40 06 01 00 00 01 45 15 01 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 41 06 01 00 00 01
match Tag-1 = { Tunnel-Type = ::PPTP, Tunnel-Medium-Type = ::IPv4 }, raw.Tunnel-Password = 0x019973851de6c557307dacd5daa599f4e26e7e

#
#  When no attribute with a tag can be decoded, there is no Tag-N group.
#
decode-pair 45 15 02 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e
match raw.Tunnel-Password = 0x029973851de6c557307dacd5daa599f4e26e7e

decode-pair 45 15 02 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 45 15 02 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e
match raw.Tunnel-Password = 0x029973851de6c557307dacd5daa599f4e26e7e, raw.Tunnel-Password = 0x029973851de6c557307dacd5daa599f4e26e7e

#
#  The Tag-N group goes where the first good attribute with that tag is, and later attributes
#  with that tag are added to it.
#
decode-pair 45 15 02 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 40 06 01 00 00 01 40 06 02 00 00 03 41 06 02 00 00 01
match raw.Tunnel-Password = 0x029973851de6c557307dacd5daa599f4e26e7e, Tag-1 = { Tunnel-Type = ::PPTP }, Tag-2 = { Tunnel-Type = ::L2TP, Tunnel-Medium-Type = ::IPv4 }

#
#  The same in a full packet.  The Tag-2 group goes before User-Name, and the later
#  Tunnel-Medium-Type is added to it.
#
decode-proto 02 01 00 3a 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 01 45 15 02 99 73 85 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 40 06 02 00 00 03 01 05 62 6f 62 41 06 02 00 00 01
match Packet-Type = ::Access-Accept, Packet-Authentication-Vector = 0x00000000000000000000000000000001, raw.Tunnel-Password = 0x029973851de6c557307dacd5daa599f4e26e7e, Tag-2 = { Tunnel-Type = ::L2TP, Tunnel-Medium-Type = ::IPv4 }, User-Name = "bob"

#
#  Tagged integers must be 4 bytes long, otherwise they are raw, and no Tag-N group is created.
#
decode-pair 40 05 01 00 00 40 06 01 00 00 01
match raw.Tunnel-Type = 0x010000, Tag-1 = { Tunnel-Type = ::PPTP }

decode-pair 40 07 01 00 00 01 00
match raw.Tunnel-Type = 0x0100000100

#
#  The encrypted data for "foo" above is 16 octets.  The first one is the length field, so the
#  password can be at most 15 octets.  These tests change the first encrypted octet, so that it
#  decrypts to a different length.  0x09 decrypts to 15, 0x16 to 16, and 0x17 to 17.
#
#  A length of 15 is allowed.  The value is "foo", followed by the padding which the encoder added.
#
decode-pair 45 15 00 99 73 09 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e
match Tunnel-Password = "foo\000D\276\223\251\304Аf\004\2741"

#
#  A length of 16 is too long.  It would include one octet which was never decrypted.
#
decode-pair 45 15 00 99 73 16 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e
match raw.Tunnel-Password = 0x009973161de6c557307dacd5daa599f4e26e7e

decode-pair 45 15 00 99 73 17 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e
match raw.Tunnel-Password = 0x009973171de6c557307dacd5daa599f4e26e7e

#
#  The same checks apply to a tagged Tunnel-Password.
#
decode-pair 45 15 01 99 73 16 1d e6 c5 57 30 7d ac d5 da a5 99 f4 e2 6e 7e 40 06 01 00 00 01
match raw.Tunnel-Password = 0x019973161de6c557307dacd5daa599f4e26e7e, Tag-1 = { Tunnel-Type = ::PPTP }
