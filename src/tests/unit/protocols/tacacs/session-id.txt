#  Test vectors for TACACS+ session IDs
#
#  When a request has no Session-Id, the encoder picks a random one.
#  rand-seed fixes the sequence of random numbers, so the session ID is
#  the same on every run.
#
proto tacacs
proto-dictionary tacacs
fuzzer-out tacacs

#
#  With no argument, the seed is 0
#
rand-seed
match 0
